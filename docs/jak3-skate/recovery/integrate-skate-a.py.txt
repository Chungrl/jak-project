(defbehavior target-skate-ground? target ()
  (and (target-skate-eligible?) (board-on-ground?)
       (= (-> self control ground-pat mode) (pat-mode ground))
       (= (-> self control mod-surface) *board-walk-mods*)))

(defbehavior target-skate-momentum target ()
  (let* ((velocity (-> self control transv-ctrl))
         (push (- (skate-stick-axis (the int (-> self control cpad lefty)))))
         (friction 1.0)
         (direction (if (< (-> velocity z) 0.0) -1.0 1.0)))
    ;; Surface friction has already been removed from this per-target working
    ;; copy; use the skate coefficient once. Gravity supplies slope acceleration.
    (set! (-> velocity z)
      (* direction 4096.0
         (skate-speed-step (/ (fabs (-> velocity z)) 4096.0)
                           (fmax push 0.0) (fmax (- push) 0.0)
                           friction 0.0 (seconds-per-frame))))
    (set! (-> velocity x)
      (* 4096.0 (skate-lateral-step (/ (-> velocity x) 4096.0)
                                  friction (seconds-per-frame))))
    (set! (-> self board transv-max) (* 4096.0 (-> *skate-physics-config* speed-limit))))
  (none))

(defbehavior target-skate-sample target ()
  (cond
    ((target-skate-eligible?)
     (set! (-> *jak3-skate-state* active?) #t)
     (skate-input-sample! *skate-input-state* *skate-input-config*
       (/ (- (the float (-> self control cpad rightx)) 128.0) 127.0)
       (/ (- 128.0 (the float (-> self control cpad righty))) 127.0)
       (seconds-per-frame))
     (cond
       ((and (target-skate-ground?) (can-jump? 'board)
             (-> *skate-input-state* event))
        (let ((trick (skate-input-consume! *skate-input-state*)))
          (when (skate-trick-launch! *skate-tricks* trick (-> *skate-input-state* event-strength))
            (set! (-> *jak3-skate-state* pop-speed)
              (skate-pop-speed (-> *skate-input-state* event-preload-time)
                               (-> *skate-input-state* event-flick-speed)))
            (set! (-> *jak3-skate-state* ground?) #f)
            (flush-trick-list (-> self board))
            ;; Retain native jump animation, event and mission notification path.
            ;; Equal min/max prevents vanilla X-hold duration from changing pop.
            (go target-board-jump (meters 1) (meters 1) #f))))
       ((and (not (board-on-ground?)) (= (-> *skate-tricks* phase) 'ground))
        (skate-trick-fall! *skate-tricks*)))
     (when (not (board-on-ground?))
       (skate-trick-air-step! *skate-tricks* (seconds-per-frame) 0.0 #f)))
    ((-> *jak3-skate-state* active?) (skate-runtime-reset!)))
  (none))

(defbehavior target-skate-contact target ()
  (when (target-skate-eligible?)
    (let ((contact? (or (board-on-ground?) (focus-test? self rail))))
      (when (and contact? (not (-> *jak3-skate-state* ground?))
                 (= (-> *skate-tricks* phase) 'airborne))
        (let* ((velocity-angle (vector-y-angle (-> self control transv)))
               (angle (/ (fabs (deg-diff (y-angle (-> self control)) velocity-angle)) 182.04445))
               ;; Fakie/switch may be parallel in either direction; sideways fails.
               (alignment (fmin angle (fabs (- 180.0 angle)))))
          (skate-trick-land! *skate-tricks*
            (-> *jak3-skate-state* incoming-impact) alignment)))
      (when contact? (skate-trick-support! *skate-tricks* 'ground (seconds-per-frame)))
      (set! (-> *jak3-skate-state* ground?) contact?)))
  (none))
